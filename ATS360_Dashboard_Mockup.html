SuiteQL — ATS Subscription Delinquency (replaces current saved search)
SELECT c.entityid AS ns_id, c.companyname AS client,
  t.tranid AS invoice_num,
  TO_CHAR(t.trandate,'Mon DD, YYYY') AS invoice_date,
  TO_CHAR(t.duedate,'Mon DD, YYYY') AS due_date,
  GREATEST(0,FLOOR(SYSDATE-t.duedate)) AS days_overdue,
  t.foreignamount AS amount, t.amountremaining AS balance_due,
  e.entityid AS account_manager
FROM transaction t
  INNER JOIN customer c ON c.id = t.entity
  LEFT JOIN employee e ON e.id = t.salesrep
WHERE t.type = 'CustInvc'
  AND t.status NOT IN ('CustInvc:B','CustInvc:C') -- B=Paid, C=Voided
  AND (t.memo LIKE '%ATS360%' OR t.custbody_ats_sub = 'T')
  AND t.memo NOT LIKE '%Implementation%' -- Exclude PS
  AND t.memo NOT LIKE '%Professional Services%'
ORDER BY days_overdue DESC, c.companyname ASC

SuiteQL — Implementation Pipeline (requires 14 custom fields on Customer record — see Excel dev brief)
SELECT c.entityid, c.companyname, c.custentity_ats_phase,
  TO_CHAR(c.custentity_ats_kickoff_date,'Mon DD, YYYY'),
  FLOOR(SYSDATE-c.custentity_ats_kickoff_date) AS days_in_phase,
  TO_CHAR(c.custentity_ats_target_golive,'Mon DD, YYYY'),
  e.entityid AS am, c.custentity_ats_brd_status, c.custentity_ats_addons
FROM customer c LEFT JOIN employee e ON e.id=c.custentity_ats_account_manager
WHERE c.custentity_ats_phase IS NOT NULL AND c.isinactive='F'
ORDER BY CASE c.custentity_ats_phase
  WHEN 'Implementing' THEN 1 WHEN 'UAT' THEN 2 WHEN 'HyperCare' THEN 3
  WHEN 'Live' THEN 4 ELSE 5 END, c.companyname

SuiteQL — Client Health (requires custom fields on Customer record)
SELECT c.companyname, c.custentity_ats_phase,
  c.custentity_ats_actively_using, c.custentity_ats_sentiment,
  c.custentity_ats_referenceable, e.entityid AS am,
  TO_CHAR(c.custentity_ats_last_meeting,'Mon YYYY'),
  c.custentity_ats_health_notes
FROM customer c LEFT JOIN employee e ON e.id=c.custentity_ats_account_manager
WHERE c.custentity_ats_phase NOT IN ('Churned','Cancelled')
  AND c.isinactive='F'
ORDER BY CASE c.custentity_ats_sentiment
  WHEN 'red' THEN 1 WHEN 'yellow' THEN 2 ELSE 3 END
